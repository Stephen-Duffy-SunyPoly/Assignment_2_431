(* non-tail-recursive *)
let rec makeList n =
  match n with
  | n when n < 0 -> failwith "Negative input is not allowed"
  | 0 -> []
  | _ -> makeList (n - 1) @ [n]

(* tail-recursive *)
let makeList n =
  let rec aux n acc =
    match n with
    | n when n < 0 -> failwith "Negative input is not allowed"
    | 0 -> acc
    | _ -> aux (n - 1) (n :: acc)
  in
  aux n []

let rec in_order tree =
  match tree with
  | Empty -> []
  | Leaf v -> [v]
  | Node (v, left, right) -> (in_order left) @ [v] @ (in_order right)

{- Find divisors -}
divisors :: Int -> [Int]
divisors n = [x | x <- [1..n-1], n `mod` x == 0]

{- Check if perfect -}
isPerfect :: Int -> Bool
isPerfect n = sum (divisors n) == n

{- Find perfect -}
perfs :: Int -> [Int]
perfs limit = [n | n <- [1..limit], isPerfect n]
